🗄️ 資料庫 SQL 查詢語句

📋 查詢所有餐點
SELECT meal_id, meal_name, prep_time
FROM McOS_meal
ORDER BY meal_id;
🍱 查詢套餐 + 包含餐點
select `c`.`comboID` AS `套餐編號`,`c`.`comboName` AS `套餐名稱`
                                    ,group_concat(concat(`i`.`ing_name`,' x',`total_ing`.`total_qty`,`i`.`unit`) separator '、') AS `總原物料需求` 
                                    from ((`a0303`.`McOS_comboMeals_new_new` `c` join (select `cd`.`comboID` AS `comboID`,`mc`.`ingID` AS `ingID`,sum((`cd`.`quantity` * `mc`.`qty`)) AS `total_qty` 
                                    from (`a0303`.`McOS_comboDetail_new_new` `cd` join `a0303`.`McOS_mealCost` `mc` on((`cd`.`mealID` = `mc`.`mealID`))) group by `cd`.`comboID`,`mc`.`ingID`) `total_ing` on((`c`.`comboID` = `total_ing`.`comboID`))) join `a0303`.`McOS_ingredients` `i` on((`total_ing`.`ingID` = `i`.`ing_id`))) group by `c`.`comboID`,`c`.`comboName`;
📋 查詢食譜步驟（JOIN 設備類型）
SELECT * FROM McOS_recipe AS r JOIN McOS_meal AS m ON r.mealID=m.meal_id ORDER BY stepOrder;
🌲 單品 BOM（餐點 → 原物料）
SELECT 
    `m`.`meal_name` AS `餐點名稱`,
    GROUP_CONCAT(CONCAT(`i`.`ing_name`, ' x ', `mc`.`qty`, `i`.`unit`) SEPARATOR '、') AS `標準配方與用量` 
  FROM ((`McOS_meal` `m` 
  JOIN `McOS_mealCost` `mc` ON ((`m`.`meal_id` = `mc`.`mealID`))) 
  JOIN `McOS_ingredients` `i` ON ((`mc`.`ingID` = `i`.`ing_id`))) 
  GROUP BY `m`.`meal_id`
🌴 套餐物料爆炸圖（二級 BOM）
/* 套餐 → 餐點 → 原物料 二級展開 */
SELECT 
            `c`.`comboID` AS `套餐編號`,
            `c`.`comboName` AS `套餐名稱`,
            
            -- 1. 串接內含單品（對應圖中的黃色漢堡圖示清單）
            GROUP_CONCAT(
            DISTINCT CONCAT(`m`.`meal_name`, ' * ', `cd`.`quantity`) 
            SEPARATOR '、'
            ) AS `內含單品`,
            
            -- 2. 串接總原物料需求（對應圖中的綠色字體）
            `total_ing`.`總原物料需求`
            
        FROM `a0303`.`McOS_comboMeals_new_new` `c`
        
        -- 連接單品明細與單品名稱表（已修正欄位為 meal_id）
        LEFT JOIN `a0303`.`McOS_comboDetail_new_new` `cd` ON `c`.`comboID` = `cd`.`comboID`
        LEFT JOIN `a0303`.`McOS_meal` `m` ON `cd`.`mealID` = `m`.`meal_id`
        
        -- 連接總原料計算子查詢
        LEFT JOIN (
            SELECT 
            `sub_cd`.`comboID`,
            GROUP_CONCAT(
                CONCAT(`i`.`ing_name`, ' x', `sub_total`.`total_qty`, `i`.`unit`) 
                SEPARATOR '、'
            ) AS `總原物料需求`
            FROM `a0303`.`McOS_comboDetail_new_new` `sub_cd`
            JOIN `a0303`.`McOS_mealCost` `mc` ON `sub_cd`.`mealID` = `mc`.`mealID`
            JOIN (
            SELECT 
                `cd2`.`comboID`, 
                `mc2`.`ingID`, 
                SUM((`cd2`.`quantity` * `mc2`.`qty`)) AS `total_qty`
            FROM `a0303`.`McOS_comboDetail_new_new` `cd2`
            JOIN `a0303`.`McOS_mealCost` `mc2` ON `cd2`.`mealID` = `mc2`.`mealID`
            GROUP BY `cd2`.`comboID`, `mc2`.`ingID`
            ) `sub_total` ON `sub_cd`.`comboID` = `sub_total`.`comboID` AND `mc`.`ingID` = `sub_total`.`ingID`
            JOIN `a0303`.`McOS_ingredients` `i` ON `mc`.`ingID` = `i`.`ing_id`
            GROUP BY `sub_cd`.`comboID`
        ) `total_ing` ON `c`.`comboID` = `total_ing`.`comboID`
        
        GROUP BY `c`.`comboID`, `c`.`comboName`, `total_ing`.`總原物料需求`
        ORDER BY `c`.`comboID`
